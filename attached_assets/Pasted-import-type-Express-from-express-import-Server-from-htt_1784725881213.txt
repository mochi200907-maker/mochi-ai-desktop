function getCachePath(key: string): string {
  const safeKey = key.replace(/[^a-zA-Z0-9]/g, "_");
  return path.join(CACHE_DIR, `weather_${safeKey}.json`);
}

function getCache(key: string): any | null {
  const cachePath = getCachePath(key);
  if (!fs.existsSync(cachePath)) return null;
  const data = JSON.parse(fs.readFileSync(cachePath, "utf8"));
  const age = Date.now() - data.timestamp;
  if (age > CACHE_DURATION) {
    try { fs.unlinkSync(cachePath); } catch {}
    return null;
  }
  return data.value;
}

function setCache(key: string, value: any): void {
  const cachePath = getCachePath(key);
  fs.writeFileSync(cachePath, JSON.stringify({ timestamp: Date.now(), value: value }));
}

// DEBOUNCE
// ============================================================================
const RECENT_REQUESTS = new Map<number, number>();
const DEBOUNCE_MS = 3000;

function isDuplicate(userId: number): boolean {
  const now = Date.now();
  const last = RECENT_REQUESTS.get(userId);
  if (last && (now - last) < DEBOUNCE_MS) {
    console.log("[DEBOUNCE] Duplicate request blocked for user:", userId);
    return true;
  }
  RECENT_REQUESTS.set(userId, now);
  return false;
}

function extractName(text: string): { action: "save" | "delete" | "none"; name: string | null } {
  const lower = text.toLowerCase();
  const deletePatterns = [
    /delete\s+(?:my\s+)?name/i, /remove\s+(?:my\s+)?name/i,
    /forget\s+(?:my\s+)?name/i, /clear\s+(?:my\s+)?name/i,
    /wala\s+na\s+ang\s+pangalan\s+ko/i,
    /burahin\s+(?:ang\s+)?pangalan\s+ko/i,
    /alisin\s+(?:ang\s+)?pangalan\s+ko/i,
  ];
  for (const pattern of deletePatterns) {
    if (pattern.test(lower)) return { action: "delete", name: null };
  }
  const savePatterns = [
    /(?:my\s+name\s+is|i\s+am|call\s+me|i\'m)\s+([a-zA-Z\s]+)/i,
    /pangalan\s+ko\s+ay\s+([a-zA-Z\s]+)/i,
    /ako\s+si\s+([a-zA-Z\s]+)/i,
    /tawagin\s+mo\s+akong\s+([a-zA-Z\s]+)/i,
  ];
  for (const pattern of savePatterns) {
    const match = lower.match(pattern);
    if (match && match[1]) {
      const name = match[1].trim().split(/\s+/)[0];
      if (name.length > 1) return { action: "save", name: name.charAt(0).toUpperCase() + name.slice(1) };
    }
  }
  return { action: "none", name: null };
}

// SPEAKER MODE DETECTION (AI-Driven + Direct Commands)
// ============================================================================
function extractSpeakerCommand(text: string): { action: "set" | "mute" | "unmute" | "none"; mode: string | null } {
  const lower = text.toLowerCase().trim();

  // MUTE commands
  const mutePatterns = [
    /\bmute\b/i, /\btahimik\b/i, /\bquiet\b/i, /\bsilent\b/i,
    /\bpatayin\s+(?:ang\s+)?(?:tunog|musika|audio)/i,
    /\bshut\s+up\b/i, /\bstop\s+(?:the\s+)?(?:sound|music|audio)/i,
    /\bwalang\s+tunog\b/i, /\boff\s+(?:ang\s+)?(?:speaker|audio)/i
  ];
  for (const pattern of mutePatterns) {
    if (pattern.test(lower)) return { action: "mute", mode: null };
  }

  // UNMUTE commands
  const unmutePatterns = [
    /\bunmute\b/i, /\bvolume\s+on\b/i, /\bturn\s+on\s+(?:ang\s+)?(?:tunog|audio)/i,
    /\bbalik\s+(?:ang\s+)?tunog\b/i, /\bsound\s+on\b/i, /\benable\s+audio\b/i,
    /\bplay\s+na\b/i, /\btunog\s+na\b/i
  ];
  for (const pattern of unmutePatterns) {
    if (pattern.test(lower)) return { action: "unmute", mode: null };
  }

  // SPEAKER MODE commands
  const leftPatterns = [
    /\bleft\s+(?:speaker|channel|audio)\b/i, /\blow\s+(?:speaker|volume)\b/i,
    /\bquiet\s+speaker\b/i, /\bsoft\s+(?:speaker|audio)\b/i,
    /\bmono\s+left\b/i, /\bleft\s+only\b/i, /\bsa\s+left\b/i,
    /\bmababang\s+tunog\b/i, /\bmahinang\s+speaker\b/i
  ];
  for (const pattern of leftPatterns) {
    if (pattern.test(lower)) return { action: "set", mode: "LEFT" };
  }

  const rightPatterns = [
    /\bright\s+(?:speaker|channel|audio)\b/i, /\bloud\s+(?:speaker|volume)\b/i,
    /\bmain\s+speaker\b/i, /\bhigh\s+volume\b/i, /\bmono\s+right\b/i,
    /\bright\s+only\b/i, /\bsa\s+right\b/i, /\bmalakas\s+na\s+tunog\b/i,
    /\bmataas\s+na\s+volume\b/i
  ];
  for (const pattern of rightPatterns) {
    if (pattern.test(lower)) return { action: "set", mode: "RIGHT" };
  }

  const bothPatterns = [
    /\bboth\s+(?:speakers|channels|audio)\b/i, /\bstereo\b/i,
    /\ball\s+speakers\b/i, /\bdual\s+speaker\b/i, /\bdalawa\b/i,
    /\bpareho\b/i, /\bfull\s+volume\b/i, /\bnormal\s+mode\b/i
  ];
  for (const pattern of bothPatterns) {
    if (pattern.test(lower)) return { action: "set", mode: "BOTH" };
  }

  return { action: "none", mode: null };
}

function detectLanguage(text: string): "en" | "fil" {
  const lower = text.toLowerCase();
  const filWords = [
    "ano", "ang", "bakit", "paano", "sino", "saan", "kailan", "kumusta", "salamat",
    "opo", "hindi", "oo", "mga", "ng", "sa", "na", "ay", "ko", "mo", "po", "lang",
    "naman", "talaga", "siguro", "pala", "din", "rin", "dito", "diyan", "doon",
    "panahon", "ulan", "mainit", "malamig", "bagyo", "araw", "gabi", "umaga",
    "tumugtog", "kanta", "musika", "tawagin", "pangalan", "tawag", "gusto", "ayaw",
    "maganda", "masaya", "malungkot", "pagod", "gutom", "uhaw", "lamig", "tahimik",
    "tunog", "mababa", "mataas", "malakas", "mahina", "pareho", "dalawa", "isa"
  ];
  const words = lower.split(/\s+/);
  let filCount = 0;
  for (const word of words) {
    if (filWords.includes(word)) filCount++;
  }
  return filCount / words.length > 0.3 ? "fil" : "en";
}

interface WeatherData {
  temperature: number;
  feelsLike: number;
  humidity: number;
  windSpeed: number;
  condition: string;
  description: string;
  city: string;
  country: string;
  uvIndex: number;
  visibility: number;
  pressure: number;
  maxTemp: number;
  minTemp: number;
  isDay: boolean;
}

async function fetchWeather(): Promise<WeatherData | null> {
  try {
    const cacheKey = "alfonso_weather";
    const cached = getCache(cacheKey);
    if (cached) return cached;

    const url = `https://wttr.in/${encodeURIComponent(WEATHER_LOCATION)}?format=j1`;
    const response = await fetch(url, {
      headers: { "User-Agent": "Mozilla/5.0", "Accept": "application/json" },
      signal: AbortSignal.timeout(15000)
    });

    if (!response.ok) return null;

    const data = await response.json();
    const current = data.current_condition[0];
    const area = data.nearest_area[0];
    const today = data.weather[0];

    const weatherData: WeatherData = {
      temperature: parseInt(current.temp_C),
      feelsLike: parseInt(current.FeelsLikeC),
      humidity: parseInt(current.humidity),
      windSpeed: parseInt(current.windspeedKmph),
      condition: current.weatherDesc[0].value,
      description: current.weatherDesc[0].value,
      city: area.areaName[0].value,
      country: area.country[0].value,
      uvIndex: parseInt(current.uvIndex),
      visibility: parseInt(current.visibility),
      pressure: parseInt(current.pressure),
      maxTemp: parseInt(today.maxtempC),
      minTemp: parseInt(today.mintempC),
      isDay: current.isdaytime === "yes"
    };

    setCache(cacheKey, weatherData);
    return weatherData;
  } catch (e: any) {
    return null;
  }
}

// EDGE TTS
// ============================================================================
async function generateEdgeTTS(text: string, outputPath: string, lang: "en" | "fil"): Promise<void> {
  return new Promise(async (resolve, reject) => {
    try {
      const tts = new MsEdgeTTS();
      const voice = lang === "en" ? "en-US-AriaNeural" : "fil-PH-BlessicaNeural";

      await tts.setMetadata(voice, OUTPUT_FORMAT.AUDIO_24KHZ_96KBITRATE_MONO_MP3);

      const { audioStream } = tts.toStream(text);
      const chunks: Buffer[] = [];

      audioStream.on("data", (chunk: Buffer) => {
        if (chunk && chunk.length > 0) chunks.push(chunk);
      });

      audioStream.on("end", () => {
        if (chunks.length === 0) return reject(new Error("Edge TTS returned empty audio"));
        const buf = Buffer.concat(chunks);
        fs.writeFileSync(outputPath, buf);
        resolve();
      });

      audioStream.on("error", (err: any) => reject(new Error("Edge TTS stream error: " + err.message)));
    } catch (err: any) {
      reject(new Error("Edge TTS setup error: " + err.message));
    }
  });
}

// PCM GENERATION - STEREO OUTPUT FOR DUAL SPEAKER (PCM5100A)
// ============================================================================
async function generatePCM(input: string, speakerMode: string = "BOTH"): Promise<Buffer> {
  const tmp = path.join(AUDIO_DIR, "raw_" + Date.now() + "_" + Math.random().toString(36).slice(2, 8) + ".pcm");

  return new Promise((resolve, reject) => {
    // Build ffmpeg filter based on speaker mode
    let channelFilter = "";
    switch (speakerMode.toUpperCase()) {
      case "LEFT":
      case "LOW":
        // Left channel only, right silent
        channelFilter = "pan=stereo|c0=c0|c1=0.0*c0";
        break;
      case "RIGHT":
      case "LOUD":
        // Right channel only, left silent
        channelFilter = "pan=stereo|c0=0.0*c0|c1=c0";
        break;
      case "MUTE":
        // Both channels silent (but we still send data structure, ESP32 will handle actual mute)
        channelFilter = "pan=stereo|c0=0.0*c0|c1=0.0*c0";
        break;
      case "BOTH":
      default:
        // Both channels same (mono to stereo)
        channelFilter = "pan=stereo|c0=c0|c1=c0";
        break;
    }

    ffmpeg(input)
      .audioFilters([
        "highpass=f=60", 
        "lowpass=f=8000",
        "aresample=" + VOICE_RATE + ":resampler=soxr:precision=28",  
        "aformat=sample_fmts=s16:channel_layouts=stereo",  // STEREO output
        channelFilter,
        "volume=1.0", 
        "dynaudnorm=p=0.95:g=15", 
        "loudnorm=I=-14:TP=-1.0:LRA=11"
      ])
      .audioCodec("pcm_s16le")
      .audioChannels(2)  // STEREO output for PCM5100A
      .audioFrequency(VOICE_RATE)
      .format("s16le")
      .on("error", (err) => reject(new Error("FFmpeg PCM error: " + err.message)))
      .on("end", () => { 
        const pcm = fs.readFileSync(tmp); 
        fs.unlinkSync(tmp); 
        resolve(pcm); 
      })
      .save(tmp);
  });
}

interface YTSearchResult { title: string; thumbnail: string; timestamp: string; url: string; }
interface YTDownloadResult { status: boolean; title: string; url: string; author: string; }

async function searchYouTube(query: string): Promise<YTSearchResult | null> {
  try {
    const searchUrl = `https://mostakim.onrender.com/mostakim/ytSearch?search=${encodeURIComponent(query)}`;
    const response = await fetch(searchUrl, { headers: { "User-Agent": "Mozilla/5.0", "Accept": "application/json" } });
    if (!response.ok) return null;
    const results: YTSearchResult[] = await response.json();
    if (!Array.isArray(results) || results.length === 0) return null;
    return results[0];
  } catch { return null; }
}

async function getDownloadUrl(youtubeUrl: string): Promise<string | null> {
  try {
    const dlUrl = `https://mostakim.onrender.com/m/ytDl?url=${encodeURIComponent(youtubeUrl)}`;
    const response = await fetch(dlUrl, { headers: { "User-Agent": "Mozilla/5.0", "Accept": "application/json" } });
    if (!response.ok) return null;
    const data: YTDownloadResult = await response.json();
    if (!data.status || !data.url) return null;
    return data.url;
  } catch { return null; }
}

async function fetchMusicUrl(query: string): Promise<string | null> {
  const searchResult = await searchYouTube(query);
  if (!searchResult) return null;
  return await getDownloadUrl(searchResult.url);
}

// STREAM AI RESPONSE PCM (16kHz — STEREO FOR DUAL SPEAKER)
// ============================================================================
async function streamPCM(ws: WebSocket, pcm: Buffer, sessionId: string, displayText?: string, speakerMode: string = "BOTH") {
  if (ws.readyState !== ws.OPEN) return;

  // Send speaker mode command FIRST before audio data
  ws.send("SPEAKER:" + speakerMode);
  await delay(100);

  const alignedLen = Math.floor(pcm.length / AI_CHUNK_SIZE) * AI_CHUNK_SIZE;
  const totalChunks = alignedLen / AI_CHUNK_SIZE;

  if (displayText && ws.readyState === ws.OPEN) {
    ws.send("TEXT:" + displayText); 
    console.log("[TEXT] Sent to ESP32 Bubble:", displayText);
    await delay(100);
  }

  ws.send("SESSION:" + sessionId);
  await delay(100);
  ws.send("PREPARE_RESPONSE:" + totalChunks);  
  await delay(300);

  let seq = 0;
  const prebufferLimit = Math.min(PREBUFFER_CHUNKS_AI, totalChunks);
  for (let i = 0; i < prebufferLimit * AI_CHUNK_SIZE; i += AI_CHUNK_SIZE) {
    if (ws.readyState !== ws.OPEN) return;
    const chunk = pcm.subarray(i, i + AI_CHUNK_SIZE);
    const packet = Buffer.allocUnsafe(2 + chunk.length);
    packet.writeUInt16BE(seq & 0xFFFF, 0);
    packet.set(chunk, 2);
    ws.send(packet, { binary: true });
    seq++;
    await delay(SEND_INTERVAL_MS_AI);  
  }

  ws.send("START_RESPONSE");
  await delay(100);

  try {
    for (let i = prebufferLimit * AI_CHUNK_SIZE; i < alignedLen; i += AI_CHUNK_SIZE) {
      if (ws.readyState !== ws.OPEN) return;
      const chunk = pcm.subarray(i, i + AI_CHUNK_SIZE);
      const packet = Buffer.allocUnsafe(2 + chunk.length);
      packet.writeUInt16BE(seq & 0xFFFF, 0);
      packet.set(chunk, 2);
      ws.send(packet, { binary: true });
      seq++;
      await delay(SEND_INTERVAL_MS_AI);  
    }
    await delay(500);
    for (let retry = 0; retry < 3; retry++) {
      if (ws.readyState === ws.OPEN) { 
        ws.send("FINISH_RESPONSE:" + sessionId); 
        await delay(200); 
      }
    }
  } catch (e: any) {
    try { ws.send("FINISH_RESPONSE:ERROR"); } catch {}
  }
}

// REAL-TIME MUSIC STREAMING (32kHz - STEREO)
// ============================================================================
async function streamMusicRealtime(ws: WebSocket, musicUrl: string, sessionId: string, speakerMode: string = "BOTH") {
  if (ws.readyState !== ws.OPEN) return;

  // Send speaker mode first
  ws.send("SPEAKER:" + speakerMode);
  await delay(100);

  return new Promise<void>((resolve, reject) => {
    const ffmpegArgs = [
      "-re",
      "-i", musicUrl,
      "-vn",
      "-af", `highpass=f=120,lowpass=f=12000,aresample=32000:resampler=soxr:precision=24,pan=stereo|c0=0.5*c0+0.5*c1|c1=0.5*c0+0.5*c1,aformat=sample_fmts=s16:channel_layouts=stereo,volume=1.2,loudnorm=I=-16:TP=-1.5:LRA=11`, 
      "-acodec", "pcm_s16le",
      "-ac", "2",  // STEREO output
      "-ar", "32000",
      "-f", "s16le",
      "pipe:1"
    ];

    const ffmpegProc = spawn("ffmpeg", ffmpegArgs, { stdio: ["ignore", "pipe", "pipe"] });

    let buffer = Buffer.alloc(0);
    let seq = 0;
    let started = false;
    let finished = false;
    let chunkCount = 0;
    let prebufferChunks: Buffer[] = [];

    ffmpegProc.on("error", (err) => {
      if (!finished) {
        finished = true;
        try { ws.send("ERROR:MUSIC_FAILED"); ws.send("FINISH_MUSIC:ERROR"); } catch {}
        reject(err);
      }
    });

    ffmpegProc.on("close", (code) => {
      if (!finished) {
        finished = true;
        if (buffer.length >= MUSIC_CHUNK_SIZE && ws.readyState === ws.OPEN) {
          const alignedLen = Math.floor(buffer.length / MUSIC_CHUNK_SIZE) * MUSIC_CHUNK_SIZE;
          for (let i = 0; i < alignedLen; i += MUSIC_CHUNK_SIZE) {
            const chunk = buffer.subarray(i, i + MUSIC_CHUNK_SIZE);
            const packet = Buffer.allocUnsafe(2 + chunk.length);
            packet.writeUInt16BE(seq & 0xFFFF, 0);
            packet.set(chunk, 2);
            try { ws.send(packet, { binary: true }); seq++; } catch {}
          }
        }
        for (let retry = 0; retry < 3; retry++) {
          if (ws.readyState === ws.OPEN) {
            try { ws.send("FINISH_MUSIC:" + sessionId); } catch {}
          }
        }
        resolve();
      }
    });

    ffmpegProc.stdout.on("data", async (data: Buffer) => {
      buffer = Buffer.concat([buffer, data]);

      if (!started) {
        while (buffer.length >= MUSIC_CHUNK_SIZE && prebufferChunks.length < PREBUFFER_CHUNKS_MUSIC) {
          const chunk = buffer.subarray(0, MUSIC_CHUNK_SIZE);
          buffer = buffer.subarray(MUSIC_CHUNK_SIZE);
          prebufferChunks.push(Buffer.from(chunk));
        }

        if (prebufferChunks.length >= PREBUFFER_CHUNKS_MUSIC) {
          started = true;
          ws.send("SESSION:" + sessionId);
          await delay(100);
          ws.send("PREPARE_MUSIC:0");  
          await delay(300);

          for (const chunk of prebufferChunks) {
            if (ws.readyState !== ws.OPEN) {
              finished = true;
              ffmpegProc.kill("SIGKILL");
              return;
            }
            const packet = Buffer.allocUnsafe(2 + chunk.length);
            packet.writeUInt16BE(seq & 0xFFFF, 0);
            packet.set(chunk, 2);
            ws.send(packet, { binary: true });
            seq++;
            chunkCount++;
            await delay(SEND_INTERVAL_MS_MUSIC);  
          }

          ws.send("START_MUSIC");
          await delay(100);
        }
        return;
      }

      while (buffer.length >= MUSIC_CHUNK_SIZE && !finished) {
        if (ws.readyState !== ws.OPEN) {
          finished = true;
          ffmpegProc.kill("SIGKILL");
          return;
        }

        const chunk = buffer.subarray(0, MUSIC_CHUNK_SIZE);
        buffer = buffer.subarray(MUSIC_CHUNK_SIZE);

        const packet = Buffer.allocUnsafe(2 + chunk.length);
        packet.writeUInt16BE(seq & 0xFFFF, 0);
        packet.set(chunk, 2);

        ws.send(packet, { binary: true });
        seq++;
        chunkCount++;

        await delay(SEND_INTERVAL_MS_MUSIC);  
      }
    });

    setTimeout(() => {
      if (!finished) {
        finished = true;
        ffmpegProc.kill("SIGKILL");
        try { ws.send("FINISH_MUSIC:" + sessionId); } catch {}
        resolve();
      }
    }, 600000);
  });
}

function delay(ms: number): Promise<void> {
  return new Promise(r => setTimeout(r, ms));
}

interface StreamingSTTSession {
  userId: number;
  ws: WebSocket;
  audioBuffer: Buffer;
  isRecording: boolean;
  sessionId: string;
}

function createSTTSession(ws: WebSocket, userId: number): StreamingSTTSession {
  return {
    userId, ws, audioBuffer: Buffer.alloc(0), isRecording: false,
    sessionId: Date.now() + "_" + Math.random().toString(36).slice(2, 8),
  };
}

async function processFinalSTT(session: StreamingSTTSession): Promise<string | null> {
  if (session.audioBuffer.length < 1600) return null;

  const tmpWav = path.join(UPLOAD_DIR, session.sessionId + "_stt.wav");
  const tmpClean = path.join(UPLOAD_DIR, session.sessionId + "_clean.wav");
  const dataLen = session.audioBuffer.length;

  try {
    const wavBuffer = Buffer.alloc(44 + dataLen);
    wavBuffer.write("RIFF", 0);
    wavBuffer.writeUInt32LE(36 + dataLen, 4);
    wavBuffer.write("WAVE", 8);
    wavBuffer.write("fmt ", 12);
    wavBuffer.writeUInt32LE(16, 16);
    wavBuffer.writeUInt16LE(1, 20);
    wavBuffer.writeUInt16LE(1, 22);
    wavBuffer.writeUInt32LE(STT_STREAM_SAMPLE_RATE, 24);
    wavBuffer.writeUInt32LE(STT_STREAM_SAMPLE_RATE * 2, 28);
    wavBuffer.writeUInt16LE(2, 32);
    wavBuffer.writeUInt16LE(16, 34);
    wavBuffer.write("data", 36);
    wavBuffer.writeUInt32LE(dataLen, 40);
    session.audioBuffer.copy(wavBuffer, 44);

    fs.writeFileSync(tmpWav, wavBuffer);

    await new Promise<void>((resolve, reject) => {
      ffmpeg(tmpWav)
        .audioFilters(["highpass=f=80", "lowpass=f=8000", "dynaudnorm=p=0.95:g=15", "afftdn=nf=-25"])
        .audioCodec("pcm_s16le")
        .audioChannels(1)
        .audioFrequency(16000)
        .on("error", (err) => {
          fs.copyFileSync(tmpWav, tmpClean);
          resolve();
        })
        .on("end", resolve)
        .save(tmpClean);
    });

    const stt = await Promise.race([
      sttClient.audio.transcriptions.create({ 
        file: fs.createReadStream(tmpClean), model: "whisper-large-v3-turbo", language: "en",
        prompt: "The user speaks English or Tagalog (Filipino)."
      }),
      new Promise<never>((_, reject) => setTimeout(() => reject(new Error("STT_TIMEOUT")), 15000))
    ]);

    const text = stt.text?.trim();
    try { fs.unlinkSync(tmpWav); } catch {}
    try { fs.unlinkSync(tmpClean); } catch {}
    return text || null;
  } catch {
    try { fs.unlinkSync(tmpWav); } catch {}
    try { fs.unlinkSync(tmpClean); } catch {}
    return null;
  }
}

interface AIAction {
  type: "chat" | "weather" | "music" | "speaker";
  text: string;
  music?: string | null;
  weather?: boolean;
  lang: "en" | "fil";
  speakerMode?: string;  // "LEFT", "RIGHT", "BOTH", "MUTE"
  shouldMute?: boolean;
  shouldUnmute?: boolean;
}

async function getAIDecision(userText: string, history: any[], savedName: string | null, currentSpeakerMode: string): Promise<AIAction> {
  const systemPrompt = `You are Mochi, a helpful voice assistant with a DUAL SPEAKER SYSTEM.
The user is in Alfonso, Cavite, Philippines.

HARDWARE:
- PCM5100A Stereo DAC
- LEFT channel = LOW speaker (quiet/background)
- RIGHT channel = LOUD speaker (main/loud)
- User can select: LEFT only, RIGHT only, BOTH, or MUTE

CRITICAL RULES:
1. You MUST respond in valid JSON ONLY. No extra text, no markdown.
2. Detect the user's language and respond in that same language.
3. If user asks about weather, set "type": "weather" and "weather": true.
4. If user wants music/song, set "type": "music" and provide "music" search query.
5. If user asks to change speaker mode (mute, unmute, left speaker, right speaker, both speakers), set "type": "speaker".
6. For speaker commands, set the appropriate "speakerMode": "LEFT", "RIGHT", "BOTH", or "MUTE".
7. Keep responses concise (1-2 sentences max).
8. Be smart about speaker selection:
   - If user says "mute" or "tahimik" → speakerMode: "MUTE", shouldMute: true
   - If user says "unmute" or "tunog" → speakerMode: "BOTH", shouldUnmute: true  
   - If user says "left speaker" or "low volume" → speakerMode: "LEFT"
   - If user says "right speaker" or "loud" → speakerMode: "RIGHT"
   - If user says "both speakers" or "stereo" → speakerMode: "BOTH"

JSON FORMAT:
{
  "type": "chat" | "weather" | "music" | "speaker",
  "text": "your response text",
  "music": "song search query or null",
  "weather": true or false,
  "lang": "en" or "fil",
  "speakerMode": "LEFT" | "RIGHT" | "BOTH" | "MUTE" or null,
  "shouldMute": true or false,
  "shouldUnmute": true or false
}
Return ONLY the JSON object.`;

  const messages: any[] = [
    { role: "system", content: systemPrompt },
    ...history.map(h => ({ role: h.role, content: h.content })),
    { role: "user", content: userText }
  ];

  const ai = await Promise.race([
    llmClient.chat.completions.create({ model: "llama-3.3-70b-versatile", messages, max_tokens: 200, temperature: 0.3 }),
    new Promise<never>((_, reject) => setTimeout(() => reject(new Error("LLM_TIMEOUT")), 10000))
  ]);

  const raw = ai.choices[0].message.content || "{}";

  try {
    const parsed = JSON.parse(raw);
    return {
      type: parsed.type || "chat",
      text: parsed.text || "Sorry, I didn't understand.",
      music: parsed.music || null,
      weather: parsed.weather === true,
      lang: parsed.lang || detectLanguage(userText),
      speakerMode: parsed.speakerMode || null,
      shouldMute: parsed.shouldMute === true,
      shouldUnmute: parsed.shouldUnmute === true
    };
  } catch {
    // Fallback: parse with regex patterns if JSON fails
    const lower = raw.toLowerCase();
    const lang = detectLanguage(userText);

    // Check for speaker commands in raw text
    const speakerCmd = extractSpeakerCommand(userText);
    if (speakerCmd.action !== "none") {
      return {
        type: "speaker",
        text: raw.replace(/[{}"]/g, "").trim(),
        music: null,
        weather: false,
        lang,
        speakerMode: speakerCmd.mode || (speakerCmd.action === "mute" ? "MUTE" : speakerCmd.action === "unmute" ? "BOTH" : currentSpeakerMode),
        shouldMute: speakerCmd.action === "mute",
        shouldUnmute: speakerCmd.action === "unmute"
      };
    }

    if (lower.includes("weather") || lower.includes("panahon")) {
      return { type: "weather", text: raw, music: null, weather: true, lang };
    }
    if (lower.includes("music") || lower.includes("song") || lower.includes("kanta")) {
      return { type: "music", text: raw, music: null, weather: false, lang };
    }
    return { type: "chat", text: raw.replace(/[{}"]/g, "").trim(), music: null, weather: false, lang };
  }
}

// PROCESS AI RESPONSE WITH DUAL SPEAKER SUPPORT
// ============================================================================
async function processAIResponse(ws: WebSocket, userText: string, userId: number, sessionId: string) {
  if (isDuplicate(userId)) { ws.send("ERROR:PROCESSING_BUSY"); return; }
  const filesToCleanup: string[] = [];

  try {
    // Check for direct speaker commands first (faster than AI for simple commands)
    const directSpeakerCmd = extractSpeakerCommand(userText);

    const nameAction = extractName(userText);
    let nameResponse = "";
    if (nameAction.action === "save" && nameAction.name) {
      await storage.saveName(userId, nameAction.name);
      nameResponse = `Nice to meet you, ${nameAction.name}!`;
    } else if (nameAction.action === "delete") {
      await storage.deleteSavedName(userId);
      nameResponse = "Name deleted.";
    }

    const history = await storage.getConversationHistory(userId);
    const savedName = await storage.getSavedName(userId);

    // Get current speaker mode from storage (if available) or default to BOTH
    let currentSpeakerMode = "BOTH";
    try {
      const savedMode = await storage.getSpeakerMode(userId);
      if (savedMode) currentSpeakerMode = savedMode;
    } catch { /* ignore */ }

    const action = await getAIDecision(userText, history, savedName, currentSpeakerMode);

    let finalText = action.text;
    let musicQuery = action.music;
    let speakerMode = action.speakerMode || currentSpeakerMode;

    // Handle speaker mode changes
    if (action.type === "speaker" || directSpeakerCmd.action !== "none") {
      // Use direct command if detected, otherwise use AI's decision
      if (directSpeakerCmd.action === "mute" || action.shouldMute) {
        speakerMode = "MUTE";
        await storage.setSpeakerMode(userId, "MUTE");
        finalText = action.lang === "fil" ? "Naka-mute na ang speaker." : "Speaker is now muted.";
      } else if (directSpeakerCmd.action === "unmute" || action.shouldUnmute) {
        speakerMode = "BOTH";
        await storage.setSpeakerMode(userId, "BOTH");
        finalText = action.lang === "fil" ? "Naka-unmute na ang speaker." : "Speaker is now unmuted.";
      } else if (directSpeakerCmd.action === "set" && directSpeakerCmd.mode) {
        speakerMode = directSpeakerCmd.mode;
        await storage.setSpeakerMode(userId, directSpeakerCmd.mode);
        const modeText = directSpeakerCmd.mode === "LEFT" ? "low speaker" : 
                        directSpeakerCmd.mode === "RIGHT" ? "loud speaker" : "both speakers";
        finalText = action.lang === "fil" 
          ? `Naka-set na ang ${modeText}.` 
          : `Speaker set to ${modeText}.`;
      } else if (action.speakerMode) {
        speakerMode = action.speakerMode;
        await storage.setSpeakerMode(userId, action.speakerMode);
      }

      // Generate TTS for speaker feedback
      const mp3 = path.join(AUDIO_DIR, sessionId + "_speaker.mp3");
      await generateEdgeTTS(finalText, mp3, action.lang);
      filesToCleanup.push(mp3);
      const pcm = await generatePCM(mp3, speakerMode);
      await streamPCM(ws, pcm, sessionId, finalText, speakerMode);

      await storage.addMessage(userId, "user", userText);
      await storage.addMessage(userId, "assistant", finalText);
      return;
    }

    if (nameResponse) {
      finalText = nameResponse;
      musicQuery = null;
    }

    if (action.type === "weather" || action.weather) {
      const weather = await fetchWeather();
      if (weather) {
        const weatherText = formatWeatherResponse(weather, action.lang);
        const mp3 = path.join(AUDIO_DIR, sessionId + "_weather.mp3");
        filesToCleanup.push(mp3);

        await generateEdgeTTS(weatherText, mp3, action.lang);
        const pcm = await generatePCM(mp3, speakerMode);
        await streamPCM(ws, pcm, sessionId, weatherText, speakerMode);

        await storage.addMessage(userId, "user", userText);
        await storage.addMessage(userId, "assistant", weatherText);
        return;
      } else {
        const failText = action.lang === "fil" ? "Pasensya na, hindi ko makuha ang panahon." : "Sorry, weather fetch failed.";
        const mp3 = path.join(AUDIO_DIR, sessionId + "_weather_fail.mp3");
        filesToCleanup.push(mp3);
        await generateEdgeTTS(failText, mp3, action.lang);
        const pcm = await generatePCM(mp3, speakerMode);
        await streamPCM(ws, pcm, sessionId, failText, speakerMode);
        return;
      }
    }

    await storage.addMessage(userId, "user", userText);
    await storage.addMessage(userId, "assistant", finalText);

    if (action.type === "music" && musicQuery) {
      const musicUrl = await fetchMusicUrl(musicQuery);
      if (!musicUrl) { 
        const notFound = action.lang === "fil" ? "Hindi ko mahanap ang kanta." : "Song not found.";
        const mp3 = path.join(AUDIO_DIR, sessionId + "_notfound.mp3");
        filesToCleanup.push(mp3);
        await generateEdgeTTS(notFound, mp3, action.lang);
        const pcm = await generatePCM(mp3, speakerMode);
        await streamPCM(ws, pcm, sessionId, notFound, speakerMode);
        return; 
      }

      if (finalText && !nameResponse) {
        const introId = sessionId + "_intro";
        const introMp3 = path.join(AUDIO_DIR, introId + ".mp3");
        await generateEdgeTTS(finalText, introMp3, action.lang);
        filesToCleanup.push(introMp3);
        const introPcm = await generatePCM(introMp3, speakerMode);
        await streamPCM(ws, introPcm, introId, finalText, speakerMode);
        await delay(1000);
      }
      await streamMusicRealtime(ws, musicUrl, sessionId + "_music", speakerMode);
    } 
    else {
      const mp3 = path.join(AUDIO_DIR, sessionId + ".mp3");
      await generateEdgeTTS(finalText, mp3, action.lang);
      filesToCleanup.push(mp3);
      const pcm = await generatePCM(mp3, speakerMode);
      await streamPCM(ws, pcm, sessionId, finalText, speakerMode);
    }
  } catch (e: any) {
    ws.send("ERROR:" + (e.message || "UNKNOWN"));
    try { ws.send("FINISH_RESPONSE:ERROR"); } catch {}
  } finally {
    for (const f of filesToCleanup) { try { fs.unlinkSync(f); } catch {} }
    setTimeout(() => RECENT_REQUESTS.delete(userId), DEBOUNCE_MS);
  }
}

// ROUTES REGISTER
// ============================================================================
export async function registerRoutes(httpServer: Server, app: Express) {
  try {
    await pushSchema();
    await verifySchema();
    console.log("[SERVER] Database ready");
  } catch (e: any) { console.error("[SERVER] DB init failed:", e.message); }

  const wss = new WebSocketServer({ server: httpServer, path: "/ws/audio", perMessageDeflate: false, maxPayload: 64 * 1024 });

  wss.on("connection", (ws: WebSocket) => {
    console.log("ESP connected — Stack-Chan Face + Dual Speaker Voice AI Mode");
    let currentUserId: number | null = null;
    let messageCount = 0;
    let sttSession: StreamingSTTSession | null = null;
    let currentSpeakerMode = "BOTH";  // Default speaker mode per connection

    ws.on("message", async (data: any, isBinary: boolean) => {
      messageCount++;
      if (!isBinary) {
        const msg = data.toString();

        // Handle ESP32 config sync
        if (msg.startsWith("CONFIG:")) {
          console.log("[ESP32 CONFIG]", msg);
          // Parse and store ESP32's current config if needed
          return;
        }

        if (msg === "READY") { 
          ws.send("STATE:IDLE"); 
        }
        else if (msg === "STREAM_START") {
          if (!currentUserId) {
            try { 
              const anon = await storage.getOrCreateUser("anon_" + Date.now()); 
              currentUserId = anon.id; 
            } catch { 
              ws.send("ERROR:DB_FAILED"); 
              return; 
            }
          }
          sttSession = createSTTSession(ws, currentUserId);
          sttSession.isRecording = true;
          ws.send("STREAM_READY");
        }
        else if (msg === "STREAM_END") {
          if (sttSession && sttSession.isRecording) {
            sttSession.isRecording = false;
            const text = await processFinalSTT(sttSession);
            if (text) {
              ws.send("STT_RESULT:" + text);
              await processAIResponse(ws, text, sttSession.userId, sttSession.sessionId);
            } else {
              ws.send("ERROR:NO_SPEECH");
              ws.send("STATE:IDLE");
            }
            sttSession = null;
          }
        }
        else if (msg.startsWith("USER:")) {
          const userName = msg.replace("USER:", "").trim();
          try {
            const user = await storage.getOrCreateUser(userName);
            currentUserId = user.id;
            // Load user's saved speaker mode
            try {
              const savedMode = await storage.getSpeakerMode(user.id);
              if (savedMode) currentSpeakerMode = savedMode;
            } catch { /* ignore */ }
            ws.send("USER_CONFIRMED:" + user.name);
          } catch { ws.send("ERROR:USER_FAILED"); }
        }
        // Handle speaker mode feedback from ESP32
        else if (msg.startsWith("SPEAKER_MODE:")) {
          console.log("[SPEAKER] ESP32 mode changed to:", msg.substring(13));
        }
        else if (msg === "MUTED" || msg === "UNMUTED") {
          console.log("[SPEAKER] ESP32 mute state:", msg);
        }
        return;
      }

      if (!sttSession || !sttSession.isRecording) return;
      const chunk = Buffer.from(data);
      const maxBytes = STT_STREAM_SAMPLE_RATE * 2 * STT_MAX_AUDIO_SECONDS;

      if (sttSession.audioBuffer.length + chunk.length > maxBytes) {
        sttSession.isRecording = false;
        const text = await processFinalSTT(sttSession);
        if (text) {
          ws.send("STT_RESULT:" + text);
          await processAIResponse(ws, text, sttSession.userId, sttSession.sessionId);
        } else {
          ws.send("ERROR:AUDIO_TOO_LONG");
          ws.send("STATE:IDLE");
        }
        sttSession = null;
        return;
      }
      sttSession.audioBuffer = Buffer.concat([sttSession.audioBuffer, chunk]);
    });

    ws.on("close", () => { 
      sttSession = null;
      currentUserId = null; 
    });

    const pingInterval = setInterval(() => {
      if (ws.readyState === ws.OPEN) ws.ping();
      else clearInterval(pingInterval);
    }, 15000);
  });

  return httpServer;
}
